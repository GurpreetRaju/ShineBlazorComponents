<div class="row">
    <div class="col">
        <FlexBox AlignItems="FlexAlign.Center">
            <Text>Current Value: @_valueColor</Text>
            <RangeDial TValue="int" Variant="ProgressVariant.Color" Size="150"
            Value="_valueColor" ValueChanged="(v) => HandleValueChanged(v, false)" />
        </FlexBox>
    </div>
    <div class="col">
        <FlexBox AlignItems="FlexAlign.Center">
            <Text>CurrentValue: @_valueGradient</Text>
            <RangeDial TValue="int" Variant="ProgressVariant.Gradient" Size="150" ShowValue="false"
                        Value="_valueGradient" ValueChanged="(v) => HandleValueChanged(v, true)">
                <Alert Color="@GetColor()" Style="padding: 0.25rem 1rem;">
                    Current Value: @_valueGradient
                </Alert>
            </RangeDial>
        </FlexBox>
    </div>
</div>

@code {
    private int _valueColor = 10;
    private int _valueGradient = 50;

    private void HandleValueChanged(int value, bool grad)
    {
        if (grad)
            _valueGradient = value;
        else
            _valueColor = value; 

        InvokeAsync(StateHasChanged);
    }

    private Color GetColor()
    {
        return _valueGradient > 66 ? Color.Danger : (_valueGradient < 33 ? Color.Success : Color.Warning);
    }
}
